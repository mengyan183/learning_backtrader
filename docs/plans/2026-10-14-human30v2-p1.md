```python
# fg_system/human30/store.py
# -*- coding: utf-8 -*-
"""Human 3.0 数据层：读写 + 原子写 + 损坏抛错 + 懒迁移 + 同日合并。

数据文件：Data/human30.json（运行态，不入库，见 .gitignore）
"""
import json
import os
from datetime import date

DATA_PATH = os.path.join(
    os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))),
    "Data", "human30.json")


def _load():
    """读数据文件。文件不存在 ⇒ 空历史（唯一允许降级的情形）。"""
    if not os.path.exists(DATA_PATH):
        return {"records": []}
    try:
        with open(DATA_PATH, encoding="utf-8") as f:
            return json.load(f)
    except Exception as e:
        # S-2（2026-10-12）：**不得**把「文件损坏」降级成「空历史」——
        # 否则下一次 _save() 会把损坏文件整个覆盖，用户打卡记录永久丢失。
        raise RuntimeError(
            "human30 数据文件损坏/不可读，拒绝按空历史处理"
            "（否则下一次写入会把它覆盖丢失）：%s（%s: %s）"
            % (DATA_PATH, type(e).__name__, e))


def _save(data):
    """原子写：tmp + fsync + os.replace，杜绝半截 JSON。"""
    d = os.path.dirname(DATA_PATH)
    if d and not os.path.exists(d):
        os.makedirs(d, exist_ok=True)
    tmp = DATA_PATH + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=1)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, DATA_PATH)


def record(mind, body, spirit, vocation, note="", when=None, market=None):
    """追加一条自评记录（同日重复打卡则覆盖当日）。返回该记录。

    ⚠️ 本步**故意照搬 v1 的「删旧 + append」写法**（characterization first），
    让 Task 4 的 C-3 测试能先红后绿。
    """
    for v in (mind, body, spirit, vocation):
        if not (0 <= v <= 100):
            raise ValueError("四象限评分必须在 0-100 之间")
    rec = {
        "date": when or date.today().isoformat(),
        "mind": float(mind), "body": float(body),
        "spirit": float(spirit), "vocation": float(vocation),
        "note": note,
    }
    if market:
        m = {}
        try:
            m["fg_index"] = round(float(market.get("fg_index")), 1)
        except (TypeError, ValueError):
            pass
        try:
            m["zone"] = int(market.get("zone"))
        except (TypeError, ValueError):
            pass
        if m:
            rec["market"] = m

    data = _load()
    data.setdefault("records", [])
    # v1 现状：同日直接删旧 + append（⚠️ 会丢 v2 字段 —— Task 4 修）
    data["records"] = [r for r in data["records"] if r.get("date") != rec["date"]]
    data["records"].append(rec)
    data["records"].sort(key=lambda r: r.get("date") or "")
    _save(data)
    return rec


def latest():
    records = _load().get("records", [])
    return records[-1] if records else None


def history(n=30):
    return _load().get("records", [])[-n:]
```

```python
def _merge_same_day(old, new):
    """同日合并：新记录覆盖同名字段，**保留 old 的所有未知字段**（C-3 修法）。"""
    if old is None:
        return dict(new)
    merged = dict(old)          # 先保留旧记录的全部字段（含 v2 的 levels/phases/...）
    for k, v in new.items():
        if isinstance(v, dict) and isinstance(merged.get(k), dict):
            sub = dict(merged[k])
            sub.update(v)
            merged[k] = sub
        else:
            merged[k] = v
    return merged
```

```python
def bucket_level(score):
    """旧记录分档映射（〔自创〕，沿用 v1 阈值）：>70→3，40..70→2，<40→1。"""

    if "phases" not in out:

    return out
```
